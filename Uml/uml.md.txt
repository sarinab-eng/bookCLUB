    %% --- LAYER 1: SERVER SIDE ---
    class BookClubServer {
        - QTcpServer* m_server
        - QList~QTcpSocket*~ m_clients
        + startServer(quint16 port) : void
        + incomingConnection(qintptr handle) : void
        - onReadyRead() : void
        - onClientDisconnected() : void
        - sendResponse(QTcpSocket* socket, QJsonObject response) : void
        
        %% Handlers
        - handleLogin(QTcpSocket* socket, QJsonObject data) : void
        - handleRegister(QTcpSocket* socket, QJsonObject data) : void
        - handlePostReview(QTcpSocket* socket, QJsonObject data) : void
        - handleCheckout(QTcpSocket* socket, QJsonObject data) : void
        - handleFetchNotifications(QTcpSocket* socket, QJsonObject data) : void
        - handleMarkNotificationRead(QTcpSocket* socket, QJsonObject data) : void
        
        %% Notification & Helpers
        + sendNotificationToPublisher(QString publisher, QString title, QString message) : void
        - saveNotificationToFile(QString username, QString title, QString message) : void
        - broadcastReviewsUpdate(QString bookId) : void
        - loadJsonArray(QString filePath) : QJsonArray
        - saveJsonArray(QString filePath, QJsonArray array) : void
    }

    %% --- LAYER 2: CLIENT NETWORK & AUTH ---
    class AuthManager {
        - QTcpSocket* m_socket
        - QString m_currentUsername
        - QString m_userRole
        
        + connectToServer(QString host, quint16 port) : void
        + login(QString username, QString password) : void
        + registerUser(QJsonObject userData) : void
        + postReview(QString bookId, int rating, QString comment) : void
        + checkoutCart(QJsonArray cartItems) : void
        + fetchNotifications() : void
        + markNotificationAsRead(QString notifId) : void
        - sendRequest(QJsonObject request) : void
        - onReadyRead() : void
        
        %% Signals
        + loginResult(bool success, QString message, QString role) : signal
        + notificationReceived(QString title, QString message) : signal
        + notificationsListReceived(QJsonArray notifications) : signal
        + reviewsUpdated(QString bookId, QJsonArray reviews) : signal
    }

    %% --- LAYER 3: CLIENT UI PAGES ---

    %% --- RELATIONSHIPS & ASSOCIATIONS ---
    BookClubServer ..> ReviewModel : Persists/Loads (reviews.json)

    BookClubServer "1" <.. "N" AuthManager : TCP Socket (JSON Protocol)
